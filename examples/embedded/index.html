管理后台如何接入 
import { mountChatWidget, createHttpChatTransport } from '@agent-runtime/chat-ui';

const chat = mountChatWidget({
  transport: createHttpChatTransport({ baseURL: '/api/agent-chat' }),
  theme: { mode: 'light', skin: 'workbench', accent: '#0758a0' },
});

// 后台退出登录时卸载；鉴权由你的服务端提供。
